cmake_minimum_required(VERSION 3.10)
project(PERA CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(pera_cpp src/cpp/Parcial3_REMR.cpp)
if(MSVC)
  target_compile_options(pera_cpp PRIVATE /W4 /utf-8)
else()
  target_compile_options(pera_cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_test(NAME self_test COMMAND pera_cpp --self-test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME check_api
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/check_api.py $<TARGET_FILE:pera_cpp>)
endif()
